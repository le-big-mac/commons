  <p class="note">The numbers and procedures are from J. Erickson, G. Nivasch and J. Xu, <i>Fusible numbers and Peano Arithmetic</i> (Logical Methods in Computer Science 18(3), 2022), and J. Xu, <i>Survey on fusible numbers</i> (2012). They proved that the fusible numbers, in their usual order, are well-ordered with order type &epsilon;<sub>0</sub>, and that Peano Arithmetic cannot prove the true statement &ldquo;For every n &isin; N there exists a smallest fusible number larger than n.&rdquo; The simple recipe is their procedure M: <i>if x &lt; 0 return &minus;x, else return M(x &minus; M(x &minus; 1))/2</i>, which gives the distance from x to the next tame number. It stops on every input, and Peano Arithmetic cannot prove that it does.</p>
  <p class="note">What is drawn: blue and orange columns hold numbers made by the paper's recipes, each of which is fusible. An empty column holds no meek number; that it holds no fusible number at all rests on the authors' conjecture that the meek numbers are all of them, which the proved procedure bears out at every one of the grid points where I ran it (see the README beside this page). The same goes for &ldquo;first in it&rdquo; and &ldquo;the next&rdquo; in the pointer's box and the table, and for &ldquo;not fusible&rdquo; in the one-number panel: they are the fuller recipe's. Where a column would take more than three tenths of a second, or the picture more than fifteen seconds, the page stops and hatches what is left. The simple recipe's gaps just under 3 are sequence A383922 of the On-Line Encyclopedia of Integer Sequences. Nothing here is new mathematics.</p>

  function setView(p, Lp) { p = Math.max(7, Math.min(6000, p)); var lo = -(2n << BigInt(p)), hi = (8n << BigInt(p)); if (Lp < lo) Lp = lo; if (Lp > hi) Lp = hi; view = { p: p, Lp: Lp }; compute(); }

  function centreOn(v, p, col) { var n = p >= v[1] ? v[0] << BigInt(p - v[1]) : v[0] >> BigInt(v[1] - p); setView(p, n - BigInt(col === undefined ? 500 : col)); }      // put the number v at column col
  function findAnchor() {                                                            // the roundest number in view, to name the ticks from
    var p = view.p, R = view.Lp + BigInt(NX), e, m; for (e = 0; e <= p; e++) { m = -((-view.Lp) >> BigInt(p - e)); if ((m << BigInt(p - e)) < R) return D(m, e); } return colA(0); }
  function compute() {
    var my = ++job, order = [], i, done = 0, B = budgetFor(view.p); anchor = findAnchor();
    if (flP !== view.p) { fl = new F.Fusible(); flP = view.p; } fl.maxDepth = 300 + 3 * view.p;      // a clean memory at each change of zoom, so the picture does not depend on where one has been before
    for (i = 0; i < NX; i++) cols[i] = -1; for (i = 0; i < NX; i++) order.push(i); order.sort(function (a, b) { return Math.abs(a - 500) - Math.abs(b - 500); });
    var began = performance.now(); stopped = false;
    (function slice() { if (my !== job) return; var t0 = performance.now(), i;
      while (done < NX && performance.now() - t0 < 14) { i = order[done++];
        if (performance.now() - began > 15000) { cols[i] = 4; stopped = true; continue; }                  // fifteen seconds for a picture: what is left is hatched
        fl.deadline = Date.now() + 300; cols[i] = fl.columnMemo(colA(i), view.p, B); fl.deadline = 0; }      // three tenths of a second for a column
      draw(); $('progress').firstChild.style.width = (100 * done / NX) + '%'; if (done < NX) setTimeout(slice, 0); else { $('progress').firstChild.style.width = '0'; describe(); if ($('lineTable').classList.contains('on')) table(); } })();
    describe();
  }

  }
  function rel(v) { var d = F.sub(v, anchor), nm = S(anchor).length <= 14 ? S(anchor) : 'A'; return d[0] === 0n ? nm : nm + (d[0] < 0n ? ' − ' : ' + ') + (d[0] < 0n ? -d[0] : d[0]) + '·2' + sup(-d[1]); }
