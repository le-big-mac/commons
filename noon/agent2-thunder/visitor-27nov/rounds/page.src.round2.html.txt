<h1>Why thunder rumbles</h1>
<p class="lead">Lightning is over in a moment, and its whole length makes its sound at once. The sound from each part
reaches you when it has had time to travel, so the stroke is read out to you from its nearest point to its
farthest. Turn your sound on, press <b>Strike</b> and watch the ring: it is the sound you are hearing at that
moment, and it marks where on the stroke that sound set out.</p>

<ul>
  <li>Drag the listener along the ground, or press where you want to stand. Close in, at half a kilometre, the thunder begins with its loudest part. Far off it comes later (about three seconds a kilometre), quieter and shorter, and usually with its loudest part somewhere in the middle. The foot of the stroke goes dim: the bending of sound in the air hides it from you.</li>
  <li><i>A bow about you</i> (it is drawn smooth): every part of it is the same distance away, so all of its sound arrives at once. One crack, the loudest thing on this page. Now tick "crooked": the crack becomes a bang of half a second or more, about a hundred times lower in pressure than the crack (the page plays it about three times quieter).</li>
  <li><i>Pointing away</i>: the stroke lies along your line of sight. Its sound is strung out over many seconds and carries about a tenth of the energy of an upright stroke, since few stretches of it face you. That is a rumble. Untick "crooked" and all that is left is a click from each end.</li>
  <li><i>A straight one</i> (drawn smooth): one crack, as the ring first touches it, and then next to nothing but a faint click from its far end. The rumble of thunder is the crookedness of lightning: tick "crooked" to give it back.</li>

<p class="small">What this is: each point of the stroke gives off one short pulse of pressure, shaped like the letter N
and five thousandths of a second long; the pulses travel at 343 metres a second, weaken as one over the distance,
lose their high notes to the air as they go, and add up. Sound also bends upward in the air, which hides the lowest
part of a far stroke; that part is drawn dim. All of this is the model of H. S. Ribner and D. Roy (1982) as
A. Glassner describes it (2000), and Glassner's article already says that a clap comes from a stretch of channel
lying on a circle about the listener. What this leaves out: the stroke is flat, in the plane of the picture; there
are no branches, no echoes from hills, no wind, and no second and third strokes down the same path. Most of this
sound lies below 200 cycles a second, which small loudspeakers hardly give: use headphones if you can. The page
squeezes loudness four to one, or the quiet thunder could not be heard beside the loud: a sound a hundred times lower
in pressure is played about three times quieter. <a href="README.md">How it was made and checked.</a></p>

  function makeRaw() {
    if (st.kind === 'stroke') return null;                               // made whole by the core, crooked already
    if (st.kind === 'straight') return [[0, 5000], [0, 0]];
    if (st.kind === 'bow') {                                              // part of a circle about the listener, 3 km off, in the sky

      return [[st.L[0] + side * 2500 * ce, 2500 * se], [st.L[0] + side * far * ce, far * se]];

  }

    else {

    }
    compute();
  }
  function compute() {
    stop();
    st.res = TH.thunder(st.pts, st.L);
    st.computed++;

  }

  function km(m) { return (m / 1000).toFixed(1) + ' km'; }

  var BOW = null;                                 // the peak of the smooth bow's crack: every point 3 km off, so that it is the same wherever the listener stands

  function drawTrace() {
    var W = trace.width, H = trace.height, r = st.res, F = Math.round(Math.max(11.5 * DPR, 0.018 * W)), mid = (H - 1.7 * F) / 2, amp = mid - 4 * DPR;
    tc.clearRect(0, 0, W, H);
    var tmax = Math.max(r.last + 0.5, 4), left = 6 * DPR, right = W - 6 * DPR, span = right - left;
    function tx(t) { return left + t / tmax * span; }
    tc.strokeStyle = '#e4dfd6'; tc.lineWidth = 1; tc.beginPath(); tc.moveTo(left, mid); tc.lineTo(right, mid); tc.stroke();
    // the pressure, drawn as its largest and smallest value in each column
    var n = r.p.length, cols = Math.floor(span), peak = r.peak || 1;
    tc.strokeStyle = '#2a5d8f'; tc.lineWidth = Math.max(1, 1.2 * DPR); tc.beginPath();
    for (var c = 0; c < cols; c++) {
      var ta = c / cols * tmax, tb = (c + 1) / cols * tmax, ia = Math.floor((ta - r.t0) * r.fs), ib = Math.ceil((tb - r.t0) * r.fs);
      if (ib < 0 || ia >= n) continue;
      var lo = 0, hi = 0;
      for (var i = Math.max(0, ia); i < Math.min(n, ib); i++) { var v = r.p[i]; if (v < lo) lo = v; if (v > hi) hi = v; }
      tc.moveTo(left + c + 0.5, mid - hi / peak * amp - 0.5); tc.lineTo(left + c + 0.5, mid - lo / peak * amp + 0.5);
    }
    tc.stroke();
    tc.fillStyle = '#6f675e'; tc.font = F + 'px Georgia, serif'; tc.textBaseline = 'top';
    var room = span / (3.2 * F), stepS = 1;                      /* as many labels as there is room for */
    while (tmax / stepS > room) stepS = stepS === 1 ? 2 : stepS === 2 ? 5 : stepS * 2;
    var lastRight = -Infinity;                                    /* a label is left out if it would touch the one before it or run off the end */
    for (var s = 0; s <= tmax; s += stepS) {
      tc.fillRect(tx(s), H - 1.6 * F, Math.max(1, DPR), 0.3 * F);
      var word = s === 0 ? 'flash' : s + ' s', wide = tc.measureText(word).width, x0 = s === 0 ? tx(0) : tx(s) - wide / 2;
      if (x0 < lastRight + 0.6 * F || x0 + wide > W - 2 * DPR) continue;
      tc.textAlign = 'left'; tc.fillText(word, x0, H - 1.2 * F); lastRight = x0 + wide;
    }
    tc.textAlign = 'left';
    if (st.playing || st.t > 0) { tc.fillStyle = '#b8862b'; tc.fillRect(tx(Math.min(st.t, tmax)) - DPR, 2 * DPR, 2 * DPR, H - 1.9 * F); }
  }

  // ---- sound
  function samples() {                             // what goes to the loudspeaker: the pressure, scaled by the rule below
    var r = st.res, n = r.p.length, out = new Float32Array(n), k = level(r.peak) / (r.peak || 1);
    for (var i = 0; i < n; i++) out[i] = r.p[i] * k;
    return out;
  }
  function buffer() {
    var r = st.res, x = samples(), buf = st.ctx.createBuffer(1, x.length, r.fs);
    buf.getChannelData(0).set(x);
    return buf;
  }
  var REF = 0.5;                                   // a peak of this size or more is played at full volume: about the smooth bow
  function level(peak) { return 0.9 * Math.min(1, Math.sqrt(Math.sqrt(peak / REF))); }     // below that, a quarter of the decibels: loudness is squeezed four to one

  function stop() {
    if (st.source) { try { st.source.onended = null; st.source.stop(); } catch (e) {} st.source = null; }
    st.playing = false; st.flash = 0;
    document.getElementById('strike').textContent = 'Strike';
  }
  function strike() {
    if (st.playing) { stop(); st.t = 0; drawScene(); drawTrace(); return; }
    var AC = window.AudioContext || window.webkitAudioContext;
    if (!st.ctx) st.ctx = new AC();
    if (st.ctx.state === 'suspended') st.ctx.resume();
    var r = st.res, skip = document.getElementById('skip').checked;
    st.offset = skip ? Math.max(0, (r.firstHeard === null ? r.first : r.firstHeard) - 0.6) : 0;        // the time since the flash at which we begin
    if (!st.ear) { st.ear = st.ctx.createAnalyser(); st.ear.fftSize = 2048; st.ear.connect(st.ctx.destination); }   /* the sound passes through a tap that the tests can read */
    var src = st.ctx.createBufferSource(); src.buffer = buffer(); src.connect(st.ear);
    var now = st.ctx.currentTime + 0.05;
    st.tFlash = now - st.offset;
    src.start(now + Math.max(0, r.t0 - st.offset), Math.max(0, st.offset - r.t0));
    src.onended = function () { st.ended++; };
    st.source = src; st.playing = true; st.strikes++; st.flash = 1; st.clock0 = performance.now();
    document.getElementById('strike').textContent = 'Stop';
    requestAnimationFrame(frame);
  }
  function frame() {
    if (!st.playing) return;
    st.t = Math.max(0, st.ctx.currentTime - st.tFlash);
    st.flash = Math.max(0, 1 - (performance.now() - st.clock0) / 350);
    if (st.t > st.res.last + 0.6) { stop(); st.t = st.res.last + 0.6; drawScene(); drawTrace(); return; }
    drawScene(); drawTrace();
    requestAnimationFrame(frame);
  }

  // ---- hands
  function capture(ev) { try { if (scene.setPointerCapture) scene.setPointerCapture(ev.pointerId); } catch (e) { /* a pointer the browser does not know: go on without */ } }
  function nearListener(ev) {
    var b = scene.getBoundingClientRect(), lx = b.left + wx(st.L[0]) / scene.width * b.width, gz = b.top + wz(0) / scene.height * b.height;
    return Math.abs(ev.clientX - lx) < 24 && ev.clientY > gz - 44;          /* on the figure or just beside it, in CSS pixels */
  }
  function onGround(ev) {
    var b = scene.getBoundingClientRect(), gz = b.top + wz(0) / scene.height * b.height;
    return ev.clientY > gz - 6;
  }
  scene.addEventListener('pointerdown', function (ev) {
    if (nearListener(ev) || onGround(ev)) { stop(); st.t = 0; st.dragging = true; moveListener(ev); capture(ev); ev.preventDefault(); return; }
    if (st.draw) { stop(); st.t = 0; st.drawing = [inSky(toWorld(ev))]; st.drawn = 0; capture(ev); ev.preventDefault(); }
    /* a press in the sky when not drawing does nothing */
  });
  function inSky(w) { return [Math.max(X0, Math.min(X1, w[0])), Math.max(0, Math.min(Z1, w[1]))]; }
  var LONGEST = 60000;                                        /* a drawn stroke stops growing at 60 km */
  function moveListener(ev) {
    var w = toWorld(ev);
    st.L = [Math.max(X0 + 300, Math.min(X1 - 300, Math.round(w[0]))), 0];
    var d = TH.distances(st.pts, st.L);
    st.res.r = d.r;                                           // enough to redraw; the sound is worked out when the hand lets go
    for (var i = 0; i < st.pts.length; i++) st.res.hidden[i] = TH.shade(Math.abs(st.pts[i][0] - st.L[0]), st.pts[i][1] - st.L[1], TH.BEND);
    say.textContent = 'Nearest point ' + km(d.nearest) + ', farthest ' + km(d.farthest) + '.';
    drawScene();
  }
  scene.addEventListener('pointermove', function (ev) {
    if (st.drawing) { var w = inSky(toWorld(ev)), l = st.drawing[st.drawing.length - 1], step = Math.sqrt((w[0] - l[0]) * (w[0] - l[0]) + (w[1] - l[1]) * (w[1] - l[1]));
      if (step > 30 && st.drawn + step <= LONGEST) { st.drawing.push(w); st.drawn += step; drawScene(); } return; }
    if (st.dragging) moveListener(ev);
  });
  function up() {
    if (st.drawing) {
      var d = st.drawing; st.drawing = null;
      if (d.length >= 2 && TH.measure(d).length > 150) { st.kind = 'own'; st.raw = d; st.seed++; rebuild(); }
      else { drawScene(); say.textContent = 'Too short to be a stroke: draw a line at least 150 metres long. The stroke is as it was.'; }
      return;
    }
    if (st.dragging) { st.dragging = false; if (st.kind === 'bow' || st.kind === 'away') rebuild(); else compute(); }
  }
  scene.addEventListener('pointerup', up); scene.addEventListener('pointercancel', up);

  function setDraw(on) { st.draw = on; document.getElementById('b-draw').setAttribute('aria-pressed', on ? 'true' : 'false'); drawScene(); }
  var CROOKED = { stroke: true, straight: false, bow: false, away: true };       /* each kind opens as its sentence in "Things to try" describes it */
  function showScene() {                         // on a wide screen: if the picture is not wholly in sight, put its top at the top
    if (getComputedStyle(scene).position === 'sticky') return;      // (on a phone it is pinned in sight, and nothing need move)
    var r = scene.getBoundingClientRect(); if (r.top < 0 || r.bottom > window.innerHeight) window.scrollBy(0, Math.round(r.top - 8));
  }
  function choose(kind) { return function () { setDraw(false); st.kind = kind; st.seed++; document.getElementById('jag').checked = CROOKED[kind]; rebuild(); showScene(); }; }
  document.getElementById('strike').addEventListener('click', strike);
  document.getElementById('b-stroke').addEventListener('click', choose('stroke'));
  document.getElementById('b-straight').addEventListener('click', choose('straight'));
  document.getElementById('b-bow').addEventListener('click', choose('bow'));
  document.getElementById('b-away').addEventListener('click', choose('away'));
  document.getElementById('b-draw').addEventListener('click', function () { stop(); st.t = 0; setDraw(!st.draw); });
  document.getElementById('jag').addEventListener('change', function () { rebuild(); showScene(); });

  // for the tests: what the page holds, and the sound it would play, without playing it
  window.THUNDERSTATE = function () {
    var r = st.res;
    return { kind: st.kind, seed: st.seed, pieces: st.pts.length - 1, listener: st.L.slice(), playing: st.playing, t: st.t, ring: C * st.t, draw: st.draw,
             first: r.first, last: r.last, nearest: r.nearest, farthest: r.farthest, peak: r.peak, tpeak: r.tpeak, samples: r.p.length, energy: r.energy,
             early: early(0.02), ends: early(0.02) + late(0.03), hidden: hiddenCount(), crooked: document.getElementById('jag').checked, firstHeard: r.firstHeard, extent: extent(),
             strikes: st.strikes, ended: st.ended, computed: st.computed, level: level(r.peak), offset: st.offset, audio: st.ctx ? st.ctx.state : 'none', say: say.textContent,
             view: [X0, X1, Z0, Z1], length: TH.measure(st.pts).length, kink: TH.measure(st.pts).kink, first3: st.pts.slice(0, 3), last1: st.pts[st.pts.length - 1] };
  };
  function early(sec) {                                       // the share of the sound's energy in its first `sec` seconds
    var r = st.res, n = r.p.length, i1 = Math.round((r.first + sec - r.t0) * r.fs), a = 0, b = 0;
    for (var i = 0; i < n; i++) { var v = r.p[i] * r.p[i]; b += v; if (i < i1) a += v; }
    return b > 0 ? a / b : 0;
  }
  function late(sec) {                                        // and in its last `sec` seconds
    var r = st.res, n = r.p.length, i0 = Math.round((r.last - sec - r.t0) * r.fs), a = 0, b = 0;
    for (var i = 0; i < n; i++) { var v = r.p[i] * r.p[i]; b += v; if (i >= i0) a += v; }
    return b > 0 ? a / b : 0;
  }
  function extent() { var e = [Infinity, -Infinity, Infinity, -Infinity]; for (var i = 0; i < st.pts.length; i++) { var q = st.pts[i]; if (q[0] < e[0]) e[0] = q[0]; if (q[0] > e[1]) e[1] = q[0]; if (q[1] < e[2]) e[2] = q[1]; if (q[1] > e[3]) e[3] = q[1]; } return e; }
  window.THUNDERSOUND = function (every) {                    // the samples handed to the loudspeaker, thinned
    var x = samples(), out = [];
    for (var i = 0; i < x.length; i += every) out.push(x[i]);
    return out;
  };
  window.THUNDEREAR = function () {                           // what is passing to the loudspeaker now, and the time since the flash
    if (!st.ear) return { t: 0, loud: 0 };
    var d = new Float32Array(st.ear.fftSize), m = 0; st.ear.getFloatTimeDomainData(d);
    for (var i = 0; i < d.length; i++) if (Math.abs(d[i]) > m) m = Math.abs(d[i]);
    return { t: st.ctx.currentTime - st.tFlash, loud: m };
  };
  window.THUNDERSEEK = function (t) { st.t = t; drawScene(); drawTrace(); };

  var fitting = 0;
  window.addEventListener('resize', function () { clearTimeout(fitting); fitting = setTimeout(function () { fit(); drawScene(); drawTrace(); }, 60); });
  fit();
  rebuild();
})();
