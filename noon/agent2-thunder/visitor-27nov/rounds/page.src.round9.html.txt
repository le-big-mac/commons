<ul>
  <li>Drag the listener along the ground, or press where you want to stand. Close in, at half a kilometre, the thunder nearly always begins with its loudest part. Far off, at 8 km, it comes later (about three seconds a kilometre), quieter and shorter, and usually with its loudest part somewhere in the middle. The foot of the stroke goes dim: the bending of sound in the air hides it from you.</li>
  <li><button class="tip" data-kind="bow">A bow about you</button> (it is drawn smooth): every part of it is the same distance away, so all of its sound arrives at once. One crack, as loud as this page plays anything. Now <button class="tip" data-kind="bow" data-jag="on">tick “crooked”</button>: the crack becomes a bang of about a third of a second to a second, something like a hundred times lower in pressure than the crack, and is played three to four times quieter.</li>
  <li><button class="tip" data-kind="away">Pointing away</button>: the stroke lies along your line of sight. Its sound is strung out over many seconds and carries about a tenth of the energy of an upright stroke heard from 3 km, where the page opens, since few stretches of it face you. That is a rumble. <button class="tip" data-kind="away" data-jag="off">Untick “crooked”</button> and all that is left is a click from each end.</li>
  <li><button class="tip" data-kind="straight">A straight one</button> (drawn smooth): one crack, as the ring first touches it, and then next to nothing, with a faint click from its far end. From far off the crack is much weaker and the click is not, so that the crack no longer stands alone: from 9 km the click has a third of the pressure of this stroke’s own crack. The rumble of thunder is the crookedness of lightning: <button class="tip" data-kind="straight" data-jag="on">tick “crooked”</button> to give it back.</li>
  <li><button class="tip" data-draw="1">Draw your own</button>, then drag across the sky.</li>

<p class="small">What this is: each point of the stroke gives off one short pulse of pressure, shaped like the letter N
and five thousandths of a second long; the pulses travel at 343 metres a second, weaken as one over the distance,
lose their high notes to the air as they go, and add up. Sound also bends upward in the air, which hides the lowest
part of a far stroke; that part is drawn dim. The model is that of H. S. Ribner and D. Roy (1982) as
A. Glassner describes it (2000), and Glassner's article already says that a clap comes from a stretch of channel
lying on a circle about the listener. Three things in it are this page's own, matched to nothing that has been checked:
how fast the air takes the high notes away, how the hidden part fades, and most of the recipe for its made-up crooked
strokes. What this leaves out: the stroke is flat, in the plane of the picture; there
are no branches, no echoes from hills, no wind, and no second and third strokes down the same path. Two things
that Glassner's program has are left out as well: small random shifts in each sound's time of flight, for the
unsteadiness of the air; and bending that changes the length of the waves and curves their paths, where here it
only hides the foot of a stroke. Most of this
sound lies below 200 cycles a second, which small loudspeakers hardly give: use headphones if you can. The page
squeezes loudness, or the quiet thunder could not be heard beside the loud: a sound a hundred times lower
in pressure is played about three times quieter. <a href="README.md">How it was made and checked.</a></p>

(function () {
  'use strict';
  var TH = window.THUNDER, C = TH.C;
  var scene = document.getElementById('scene'), sc = scene.getContext('2d');
  var trace = document.getElementById('trace'), tc = trace.getContext('2d');
  var say = document.getElementById('say');
  // the world shown: x from X0 to X1 metres along the ground, z from Z0 up
  var X0 = -2000, X1 = 10000, Z0 = -350, Z1 = 5850;
  var st = { seed: 3, kind: 'stroke', raw: null, pts: null, L: [3000, 0], res: null, draw: false, drawing: null, dragging: false,
             playing: false, t: 0, tFlash: 0, offset: 0, source: null, ctx: null, ear: null, drawn: 0, strikes: 0, flash: 0, ended: 0, computed: 0 };

  var DPR = 1, U = 1, FONT = 22;                    // device pixels to a CSS pixel; drawing scale; size of lettering, in canvas pixels
  function fit() {                                  // both drawings take the width they are given, and their lettering stays readable
    DPR = Math.max(1, Math.min(3, window.devicePixelRatio || 1));
    var w = Math.max(200, Math.round(scene.clientWidth * DPR));
    U = Math.max(w / 1200, 0.55 * DPR); FONT = Math.round(Math.max(11.5 * DPR, 20 * w / 1200));
    var mpp = (X1 - X0) / w;                        // metres to a canvas pixel, the same up as along, so that the ring is round
    Z0 = -Math.max(350, 1.7 * FONT * mpp);          // the strip of ground is deep enough for its lettering
    scene.width = w; scene.height = Math.round((Z1 - Z0) / mpp); scene.style.height = (scene.height / DPR) + 'px';
    trace.width = Math.max(200, Math.round(trace.clientWidth * DPR)); trace.height = Math.round(118 * DPR);
  }
  function wx(x) { return (x - X0) / (X1 - X0) * scene.width; }
  function wz(z) { return scene.height - (z - Z0) / (Z1 - Z0) * scene.height; }
  function toWorld(ev) {
    var b = scene.getBoundingClientRect();
    return [X0 + (ev.clientX - b.left) / b.width * (X1 - X0), Z0 + (1 - (ev.clientY - b.top) / b.height) * (Z1 - Z0)];
  }

  function makeRaw() {
    if (st.kind === 'stroke') return null;                               // made whole by the core, crooked already
    if (st.kind === 'straight') return [[0, 5000], [0, 0]];
    if (st.kind === 'bow') {                                              // part of a circle about the listener, 3 km off, in the sky
      var R = 3000; return TH.arc(st.L[0], 0, R, 150, 95);
    }
    if (st.kind === 'away') {                                             // along the line of sight, 38 degrees up, from 2.5 km out as far as the picture allows (8 km at most)
      var ce = Math.cos(38 * Math.PI / 180), se = Math.sin(38 * Math.PI / 180);
      var right = (X1 - 150 - st.L[0]) / ce, left = (st.L[0] - X0 - 150) / ce, side = right >= left ? 1 : -1, far = Math.min(8000, Math.max(right, left), 4900 / se);
      return [[st.L[0] + side * 2500 * ce, 2500 * se], [st.L[0] + side * far * ce, far * se]];
    }
    return st.raw;
  }
  function rebuild() {
    var jagOn = document.getElementById('jag').checked;
    if (st.kind === 'stroke') st.pts = TH.stroke(st.seed, { kink: jagOn ? undefined : 0 });
    else {
      st.raw = makeRaw();
      st.pts = jagOn ? TH.jag(st.raw, st.seed * 31 + 5) : (st.kind === 'bow' ? st.raw : TH.resample(st.raw, TH.STEP));   /* the bow is cut into its pieces already, each end on the circle */
    }
    compute();
  }
  function compute() {
    stop();
    st.res = TH.thunder(st.pts, st.L);
    st.computed++;
    st.t = 0;
    describe(); drawScene(); drawTrace();
  }

  function km(m) { return (m / 1000).toFixed(1) + ' km'; }
  /* A word of lettering in a picture is drawn, and put on record, by this one function, so that what the tests read is what was drawn
     (the referee made a copy of the page whose ground read 1, 3, 5, 7, 9 km while a record kept beside the drawing said 0, 2, 4, 6, 8). */
  function letter(c, list, word, x, y, at, tick, size) {
    c.fillText(word, x, y);
    list.push({ word: word, at: at, tick: tick, x: x, y: y, w: c.measureText(word).width, h: size, font: c.font, fill: c.fillStyle, across: c.canvas.width, align: c.textAlign, base: c.textBaseline });
  }
  function hiddenCount() { var n = 0, h = st.res.hidden; for (var i = 0; i < h.length; i++) if (h[i] < 0.5) n++; return n; }
  function shown(x) { return x < 9.95 ? Number(x.toFixed(1)) : Number(x.toPrecision(2)); }      // a figure as times() prints it: the fourth root is taken of that, so that a reader who takes it gets the figure printed beside it (the referee found 68 sentences of 659 one out in the last place)
  function times(x) { return x < 9.95 ? x.toFixed(1) : Number(x.toPrecision(2)).toLocaleString('en-US'); }      // "3.4 times", "130 times", "1,500 times"
  /* How loud, in words, from the peak alone: against the crack of the smooth bow (REF), which is also what plays at full volume.
     Within 6% of it (half a decibel) a sound is called as loud; q, how many times quieter it is played, is the fourth root of rel. */
  function loudness(peak) {
    if (!(peak > 0)) return '.';                    // (silence has its own sentence; this is never reached on the page)
    var rel = REF / peak, CRACK = 'the crack that \u201cA bow about you\u201d makes';
    return rel < 1.06 && rel > 1 / 1.06 ? ', as loud as this page plays anything.'
      : rel <= 1 / 1.06 ? ', with about ' + times(1 / rel) + ' times the pressure of ' + CRACK + '; the page plays it no louder than that crack.'
      : ', with about ' + times(rel) + ' times less pressure than ' + CRACK + '; ' + (times(Math.pow(shown(rel), 0.25)) === '1.0' ? 'the page plays it very nearly as loud.' : 'it is played about ' + times(Math.pow(shown(rel), 0.25)) + ' times quieter (loudness is squeezed: see the foot of the page).');
  }
  /* The same in short, kept beside Strike on a phone: which stroke, crooked or smooth, how long it lasts and how loud it is. */
  var NAME = { stroke: 'An upright stroke', straight: 'A straight one', bow: 'A bow about you', away: 'Pointing away', own: 'Your own stroke' };
  function which() { return NAME[st.kind] + (document.getElementById('jag').checked ? ', crooked' : ', smooth') + '. '; }
  function loudShort(peak) {
    if (!(peak > 0)) return '';
    var rel = REF / peak;
    return rel < 1.06 && rel > 1 / 1.06 ? 'as loud as this page plays anything.'
      : rel <= 1 / 1.06 ? 'about ' + times(1 / rel) + ' times the pressure of the bow\u2019s crack.'
      : 'about ' + times(rel) + ' times less pressure than ' + (st.kind === 'bow' ? 'when smooth.' : 'the bow\u2019s crack.');      /* (of a crooked bow a visitor asked: which bow?) */
  }
  var brief = document.getElementById('brief');
  function describe() {
    var r = st.res, len = TH.measure(st.pts).length, fh = r.firstHeard;
    if (fh === null) { say.textContent = 'The stroke is ' + km(len) + ' long, and nearly all of it lies in the shadow that the bending of sound makes: by this model you would hear next to nothing.';
      brief.textContent = which() + 'Next to nothing to hear from here.'; return; }
    var dur = r.last - fh, lasts = dur < 0.1 ? Math.round(dur * 1000) + ' thousandths of a second' : dur.toFixed(1) + ' s';
    var loud = loudness(r.peak);
    say.innerHTML = 'The stroke is ' + km(len) + ' long. The first sound comes <b>' + fh.toFixed(1) + ' s</b> after the flash, from <b>' + km(fh * C)
      + '</b> away; the last from ' + km(r.farthest) + ', so the thunder lasts <b>' + lasts + '</b>. It is loudest ' + r.tpeak.toFixed(1) + ' s after the flash'
      + loud.replace(/about ([\d.,]+ times)/, 'about <b>$1</b>')
      + (hiddenCount() > 0 ? ' The dim part of the stroke is more than half hidden from you by the bending of sound.' : '');
    var into = (Number(r.tpeak.toFixed(1)) - Number(fh.toFixed(1))).toFixed(1);          /* how far into the thunder its loudest moment is, from the two figures as the sentence prints them */
    brief.textContent = (which() + 'First sound ' + fh.toFixed(1) + ' s, from ' + km(fh * C) + '; lasts ' + lasts + (dur < 0.1 ? '' : into === '0.0' ? ', loudest at its start' : ', loudest ' + into + ' s in') + '; ' + loudShort(r.peak))
      .replace(/(\d) (s|km|times|thousandths)\b/g, '$1\u00a0$2').replace(/(\d\u00a0s) in\b/g, '$1\u00a0in');      /* a figure is not parted from its unit at a line's end (a visitor: "lasts 7.7 / s", "loudest 0.4 s / in") */
  }

  function drawScene() {
    var W = scene.width, H = scene.height, r = st.res;
    sc.clearRect(0, 0, W, H);
    var g = sc.createLinearGradient(0, 0, 0, H); g.addColorStop(0, '#141a27'); g.addColorStop(1, '#2a3246');
    sc.fillStyle = g; sc.fillRect(0, 0, W, H);
    if (st.flash > 0) { sc.fillStyle = 'rgba(255,255,240,' + (0.35 * st.flash).toFixed(3) + ')'; sc.fillRect(0, 0, W, H); }
    // the cloud base and the ground
    sc.fillStyle = 'rgba(200,205,215,0.10)'; sc.fillRect(0, 0, W, wz(5000));
    sc.fillStyle = '#1b2118'; sc.fillRect(0, wz(0), W, H - wz(0));
    sc.fillStyle = 'rgba(244,239,226,0.62)'; sc.font = FONT + 'px Georgia, serif'; sc.textBaseline = 'top';
    st.words = [];                                             /* the picture's other words, with where they stand */
    st.marks = [];                                             /* each mark's lettering, with the place in the world where its tick is drawn: for the tests */
    for (var k = 0; k <= 8; k += 2) { var x = wx(k * 1000); sc.fillRect(x, wz(0), 1.5 * U, 0.4 * FONT); letter(sc, st.marks, k + ' km', x + 0.25 * FONT, wz(0) + 0.3 * FONT, X0 + x / scene.width * (X1 - X0), x, FONT); }
    sc.fillStyle = 'rgba(244,239,226,0.45)'; letter(sc, st.words, 'cloud', 0.5 * FONT, wz(5000) - 1.25 * FONT, null, null, FONT);
    // the ring: the sound being heard now set out from points this far away
    if (st.playing || st.t > 0) {
      var R = Math.max(0, C * st.t);
      sc.strokeStyle = 'rgba(230,180,80,0.9)'; sc.lineWidth = 2.5 * U; sc.beginPath();
      sc.ellipse(wx(st.L[0]), wz(st.L[1]), R / (X1 - X0) * W, R / (Z1 - Z0) * H, 0, Math.PI, 2 * Math.PI); sc.stroke();
    }
    // the stroke
    var p = st.pts;
    sc.lineWidth = (st.flash > 0 ? 4 : 2.5) * U; sc.lineJoin = 'round';
    for (var tone = 0; tone < 2; tone++) {                 /* first the parts heard, then, dimmer, the parts hidden by the bending of sound */
      sc.strokeStyle = tone === 0 ? (st.flash > 0 ? '#ffffff' : 'rgba(244,239,226,0.68)') : (st.flash > 0 ? 'rgba(255,255,255,0.75)' : 'rgba(244,239,226,0.22)');
      sc.beginPath(); var pen = false;
      for (var i = 0; i + 1 < p.length; i++) {
        var dim = 0.5 * (r.hidden[i] + r.hidden[i + 1]) < 0.5;
        if ((tone === 1) !== dim) { pen = false; continue; }
        if (!pen) { sc.moveTo(wx(p[i][0]), wz(p[i][1])); pen = true; }
        sc.lineTo(wx(p[i + 1][0]), wz(p[i + 1][1]));
      }
      sc.stroke();
    }
    // the part being heard: what the ring has crossed in the last twentieth of a second
    if (st.playing || st.t > 0) {
      var lo = C * (st.t - 0.05), hi = C * st.t;
      sc.fillStyle = '#ffd98a';
      for (var j = 0; j < p.length; j++) if (r.r[j] >= lo && r.r[j] <= hi) { sc.beginPath(); sc.arc(wx(p[j][0]), wz(p[j][1]), 6 * U, 0, 6.2832); sc.fill(); }
    }
    if (st.drawing && st.drawing.length > 1) {
      sc.strokeStyle = '#ffd98a'; sc.lineWidth = 3 * U; sc.beginPath();
      for (var d = 0; d < st.drawing.length; d++) { if (d === 0) sc.moveTo(wx(st.drawing[d][0]), wz(st.drawing[d][1])); else sc.lineTo(wx(st.drawing[d][0]), wz(st.drawing[d][1])); }
      sc.stroke();
    }
    // the listener
    var lx = wx(st.L[0]), lz = wz(0);
    sc.fillStyle = '#e6b450'; sc.beginPath(); sc.arc(lx, lz - 26 * U, 9 * U, 0, 6.2832); sc.fill();
    sc.fillRect(lx - 3 * U, lz - 18 * U, 6 * U, 18 * U);
    var flip = lx > W - 5.5 * FONT;
    sc.fillStyle = 'rgba(244,239,226,0.85)'; sc.textBaseline = 'bottom'; sc.textAlign = flip ? 'right' : 'left';
    letter(sc, st.words, 'you (drag)', lx + (flip ? -14 : 14) * U, lz - 12 * U, null, lx, FONT); sc.textAlign = 'left';
    if (st.draw) { sc.fillStyle = 'rgba(255,217,138,0.95)'; sc.textBaseline = 'top'; letter(sc, st.words, 'Drag across the sky to draw', 0.5 * FONT, wz(5000) + 0.4 * FONT, null, null, FONT); }
  }

  function drawTrace() {
    var W = trace.width, H = trace.height, r = st.res, F = Math.round(Math.max(11.5 * DPR, 0.018 * W)), mid = (H - 1.7 * F) / 2, amp = mid - 4 * DPR;
    tc.clearRect(0, 0, W, H);
    var tmax = Math.max(r.last + 0.5, 4), left = 6 * DPR, right = W - 6 * DPR, span = right - left;
    function tx(t) { return left + t / tmax * span; }
    tc.strokeStyle = '#e4dfd6'; tc.lineWidth = 1; tc.beginPath(); tc.moveTo(left, mid); tc.lineTo(right, mid); tc.stroke();
    // the pressure, drawn as its largest and smallest value in each column
    var n = r.p.length, cols = Math.floor(span), peak = r.peak || 1;
    tc.strokeStyle = '#2a5d8f'; tc.lineWidth = Math.max(1, 1.2 * DPR); tc.beginPath();
    for (var c = 0; c < cols; c++) {
      var ta = c / cols * tmax, tb = (c + 1) / cols * tmax, ia = Math.floor((ta - r.t0) * r.fs), ib = Math.ceil((tb - r.t0) * r.fs);
      if (ib < 0 || ia >= n) continue;
      var lo = 0, hi = 0;
      for (var i = Math.max(0, ia); i < Math.min(n, ib); i++) { var v = r.p[i]; if (v < lo) lo = v; if (v > hi) hi = v; }
      tc.moveTo(left + c + 0.5, mid - hi / peak * amp - 0.5); tc.lineTo(left + c + 0.5, mid - lo / peak * amp + 0.5);
    }
    tc.stroke();
    tc.fillStyle = '#6f675e'; tc.font = F + 'px Georgia, serif'; tc.textBaseline = 'top';
    var room = span / (3.2 * F), stepS = 1;                      /* as many labels as there is room for */
    while (tmax / stepS > room) stepS = stepS === 1 ? 2 : stepS === 2 ? 5 : stepS * 2;
    var lastRight = -Infinity;                                    /* a label is left out if it would touch the one before it or run off the end */
    st.ticks = [];                                                /* each label, with the time at the place where its tick is drawn: for the tests */
    for (var s = 0; s <= tmax; s += stepS) {
      tc.fillRect(tx(s), H - 1.6 * F, Math.max(1, DPR), 0.3 * F);
      var word = s === 0 ? 'flash' : s + ' s', wide = tc.measureText(word).width, x0 = s === 0 ? tx(0) : tx(s) - wide / 2;
      if (x0 < lastRight + 0.6 * F || x0 + wide > W - 2 * DPR) continue;
      tc.textAlign = 'left'; letter(tc, st.ticks, word, x0, H - 1.2 * F, (tx(s) - left) / span * tmax, tx(s), F); lastRight = x0 + wide;
    }
    tc.textAlign = 'left';
    if (st.playing || st.t > 0) { tc.fillStyle = '#b8862b'; tc.fillRect(tx(Math.min(st.t, tmax)) - DPR, 2 * DPR, 2 * DPR, H - 1.9 * F); }
  }

  // ---- sound
  function samples() {                             // what goes to the loudspeaker: the pressure, scaled by the rule below
    var r = st.res, n = r.p.length, out = new Float32Array(n), k = level(r.peak) / (r.peak || 1);
    for (var i = 0; i < n; i++) out[i] = r.p[i] * k;
    return out;
  }
  function buffer() {
    var r = st.res, x = samples(), buf = st.ctx.createBuffer(1, x.length, r.fs);
    buf.getChannelData(0).set(x);
    return buf;
  }
  /* The yardstick: the peak of the crack that "A bow about you" makes when smooth, every point of it 3 km from the listener.  A peak of
     this size or more is played at full volume, and the sentence under the picture sets every thunder against it.  (Until 27 November
     full volume was a peak of 0.5, and the crack stood 1.44 times above it: then the two figures in the sentence, one against the crack
     and one against full volume, did not go together as "four to one", and a visitor with a calculator found it.) */
  var REF = TH.thunder(TH.arc(0, 0, 3000, 150, 95), [0, 0]).peak;
  function level(peak) { return 0.9 * Math.min(1, Math.sqrt(Math.sqrt(peak / REF))); }     // below that, a quarter of the decibels: loudness is squeezed four to one

  function stop() {
    if (st.source) { try { st.source.onended = null; st.source.stop(); } catch (e) {} st.source = null; }
    st.playing = false; st.flash = 0;
    document.getElementById('strike').textContent = 'Strike';
  }
  function strike() {
    if (st.playing) { stop(); st.t = 0; drawScene(); drawTrace(); return; }
    var AC = window.AudioContext || window.webkitAudioContext;
    if (!st.ctx) st.ctx = new AC();
    if (st.ctx.state === 'suspended') st.ctx.resume();
    var r = st.res, skip = document.getElementById('skip').checked;
    st.offset = skip ? Math.max(0, (r.firstHeard === null ? r.first : r.firstHeard) - 0.6) : 0;        // the time since the flash at which we begin
    if (!st.ear) { st.ear = st.ctx.createAnalyser(); st.ear.fftSize = 2048; st.ear.connect(st.ctx.destination); }   /* the sound passes through a tap that the tests can read */
    var src = st.ctx.createBufferSource(); src.buffer = buffer(); src.connect(st.ear);
    var now = st.ctx.currentTime + 0.05;
    st.tFlash = now - st.offset;
    src.start(now + Math.max(0, r.t0 - st.offset), Math.max(0, st.offset - r.t0));
    src.onended = function () { st.ended++; };
    st.source = src; st.playing = true; st.strikes++; st.flash = 1; st.clock0 = performance.now();
    document.getElementById('strike').textContent = 'Stop';
    requestAnimationFrame(frame);
  }
  function frame() {
    if (!st.playing) return;
    st.t = Math.max(0, st.ctx.currentTime - st.tFlash);
    st.flash = Math.max(0, 1 - (performance.now() - st.clock0) / 350);
    if (st.t > st.res.last + 0.6) { stop(); st.t = st.res.last + 0.6; drawScene(); drawTrace(); return; }
    drawScene(); drawTrace();
    requestAnimationFrame(frame);
  }

  // ---- hands
  function capture(ev) { try { if (scene.setPointerCapture) scene.setPointerCapture(ev.pointerId); } catch (e) { /* a pointer the browser does not know: go on without */ } }
  function nearListener(ev) {
    var b = scene.getBoundingClientRect(), lx = b.left + wx(st.L[0]) / scene.width * b.width, gz = b.top + wz(0) / scene.height * b.height;
    return Math.abs(ev.clientX - lx) < 24 && ev.clientY > gz - 44;          /* on the figure or just beside it, in CSS pixels */
  }
  function onGround(ev) {
    var b = scene.getBoundingClientRect(), gz = b.top + wz(0) / scene.height * b.height;
    return ev.clientY > gz - 6;
  }
  /* A browser on a touch screen may hand a tap to a neighbour of the thing under the finger.  A tap on the picture's ground within a finger's
     breadth of Strike went to Strike (27 November: a visitor's quick taps moved nothing, and played); and once the picture answered clicks, a tap
     on Strike within ten pixels of its corners went to the picture (found by the page's test).  The pointer events still carry the finger's own
     place, so the page goes by that: the picture answers a finger that is on the picture, and Strike a finger that is on Strike, whichever of
     the two the browser chose.  A tap in the gap between them never moves the listener. */
  var down = null, strikeBtn = document.getElementById('strike');
  strikeBtn.style.minWidth = Math.ceil(strikeBtn.getBoundingClientRect().width) + 'px';      /* "Stop" is a shorter word: the button keeps the width of "Strike", so that nothing beside it shifts while the thunder plays */
  function within(el, pt, g) { var r = el.getBoundingClientRect(); g = g || 0; return pt.x >= r.left - g && pt.x <= r.right + g && pt.y >= r.top - g && pt.y <= r.bottom + g; }
  document.addEventListener('pointerdown', function (ev) {
    down = { x: ev.clientX, y: ev.clientY, t: ev.timeStamp };
    if (ev.target !== scene && within(scene, down)) sceneDown(ev);          /* handed to a neighbour, but the finger is on the picture */
  }, true);
  scene.addEventListener('pointerdown', sceneDown);
  function sceneDown(ev) {
    if (!within(scene, { x: ev.clientX, y: ev.clientY })) return;           /* handed to the picture, but the finger is not on it */
    if (nearListener(ev) || onGround(ev)) { stop(); st.t = 0; st.dragging = true; moveListener(ev); capture(ev); ev.preventDefault(); return; }
    if (st.draw) { stop(); st.t = 0; st.drawing = [inSky(toWorld(ev))]; st.drawn = 0; capture(ev); ev.preventDefault(); }
    /* a press in the sky when not drawing does nothing */
  }
  function inSky(w) { return [Math.max(X0, Math.min(X1, w[0])), Math.max(0, Math.min(Z1, w[1]))]; }
  var LONGEST = 60000;                                        /* a drawn stroke stops growing at 60 km */
  function moveListener(ev) {
    var w = toWorld(ev);
    st.L = [Math.max(X0 + 300, Math.min(X1 - 300, Math.round(w[0]))), 0];
    var d = TH.distances(st.pts, st.L);
    st.res.r = d.r;                                           // enough to redraw; the sound is worked out when the hand lets go
    for (var i = 0; i < st.pts.length; i++) st.res.hidden[i] = TH.shade(Math.abs(st.pts[i][0] - st.L[0]), st.pts[i][1] - st.L[1], TH.BEND);
    say.textContent = 'Nearest point ' + km(d.nearest) + ', farthest ' + km(d.farthest) + '.'; brief.textContent = say.textContent.replace(/(\d) km/g, '$1\u00a0km');
    drawScene();
  }
  scene.addEventListener('pointermove', function (ev) {
    if (st.drawing) { var w = inSky(toWorld(ev)), l = st.drawing[st.drawing.length - 1], step = Math.sqrt((w[0] - l[0]) * (w[0] - l[0]) + (w[1] - l[1]) * (w[1] - l[1]));
      if (step > 30 && st.drawn + step <= LONGEST) { st.drawing.push(w); st.drawn += step; drawScene(); } return; }
    if (st.dragging) moveListener(ev);
  });
  function up() {
    if (st.drawing) {
      var d = st.drawing; st.drawing = null;
      if (d.length >= 2 && TH.measure(d).length > 150) { st.kind = 'own'; st.raw = d; st.seed++; rebuild(); }
      else { drawScene(); say.textContent = 'Too short to be a stroke: draw a line at least 150 metres long. The stroke is as it was.'; }
      return;
    }
    if (st.dragging) { st.dragging = false; if (st.kind === 'bow' || st.kind === 'away') rebuild(); else compute(); }
  }
  scene.addEventListener('pointerup', up); scene.addEventListener('pointercancel', up);
  scene.addEventListener('click', function () { if (down && within(strikeBtn, down, 1)) strike(); });      /* the finger was on Strike (a pixel of grace: half a pixel above its box the browser already counts the finger as on it: the referee) */

  function setDraw(on) { st.draw = on; document.getElementById('b-draw').setAttribute('aria-pressed', on ? 'true' : 'false'); scene.classList.toggle('drawing', on); drawScene(); }
  var CROOKED = { stroke: true, straight: false, bow: false, away: true };       /* each kind opens as its sentence in "Things to try" describes it */
  function showScene() {                         // on a wide screen: if the picture is not wholly in sight, put its top at the top
    if (getComputedStyle(document.getElementById('pin')).position === 'sticky') return;      // (on a phone it is pinned in sight, and nothing need move)
    var r = scene.getBoundingClientRect(); if (r.top < 0 || r.bottom > window.innerHeight) window.scrollBy(0, Math.round(r.top - 8));
  }
  function choose(kind, crooked) { return function () { setDraw(false); st.kind = kind; st.seed++; document.getElementById('jag').checked = crooked === undefined ? CROOKED[kind] : crooked; rebuild(); showScene(); }; }
  strikeBtn.addEventListener('click', function (ev) {      /* not if this click is the end of a tap whose finger went down on the picture, a second and a half ago at most; a press from the keyboard has no finger, and detail 0 */
    if (ev.detail > 0 && down && within(scene, down) && ev.timeStamp - down.t < 1500) return;
    strike(); });
  document.getElementById('b-stroke').addEventListener('click', choose('stroke'));
  document.getElementById('b-straight').addEventListener('click', choose('straight'));
  document.getElementById('b-bow').addEventListener('click', choose('bow'));
  document.getElementById('b-away').addEventListener('click', choose('away'));
  document.getElementById('b-draw').addEventListener('click', function () { stop(); st.t = 0; setDraw(!st.draw); });
  document.getElementById('jag').addEventListener('change', function () { rebuild(); showScene(); });
  [].forEach.call(document.querySelectorAll('button.tip'), function (b) {      // the same as the buttons and the box above, pressed from inside a tip
    b.addEventListener('click', function () {
      var kind = b.getAttribute('data-kind'), jag = b.getAttribute('data-jag'), j = document.getElementById('jag'), want = jag === 'on';
      if (b.getAttribute('data-draw')) { stop(); st.t = 0; setDraw(true); showScene(); }
      else if (!jag) choose(kind)();
      else if (st.kind !== kind) choose(kind, want)();          /* "tick crooked" in a tip is about that tip's stroke: if another is showing, choose it first (the referee) */
      else { if (j.checked !== want) { j.checked = want; rebuild(); } else if (st.playing) strike(); showScene(); }      /* (already so, and playing: stop, so that Strike reads Strike, as "press Strike after each" supposes: the referee) */
    });
  });

  // for the tests: what the page holds, and the sound it would play, without playing it
  window.THUNDERLOUD = function (peak) { return loudness(peak); }; window.THUNDERLOUDSHORT = function (peak) { return loudShort(peak); };      // the page's own words for a given peak, for the tests
  window.THUNDERREF = function () { return REF; };
  window.THUNDERSTATE = function () {
    var r = st.res;
    return { kind: st.kind, seed: st.seed, pieces: st.pts.length - 1, listener: st.L.slice(), playing: st.playing, t: st.t, ring: C * st.t, draw: st.draw,
             first: r.first, last: r.last, nearest: r.nearest, farthest: r.farthest, peak: r.peak, tpeak: r.tpeak, samples: r.p.length, energy: r.energy,
             early: early(0.02), ends: early(0.02) + late(0.03), hidden: hiddenCount(), crooked: document.getElementById('jag').checked, firstHeard: r.firstHeard, extent: extent(),
             strikes: st.strikes, ended: st.ended, computed: st.computed, level: level(r.peak), offset: st.offset, audio: st.ctx ? st.ctx.state : 'none', say: say.textContent, brief: brief.textContent, marks: st.marks, ticks: st.ticks, words: st.words, lines: [wz(5000), wz(0), scene.height], sayHTML: say.innerHTML,
             view: [X0, X1, Z0, Z1], length: TH.measure(st.pts).length, kink: TH.measure(st.pts).kink, first3: st.pts.slice(0, 3), last1: st.pts[st.pts.length - 1] };
  };
  function early(sec) {                                       // the share of the sound's energy in its first `sec` seconds
    var r = st.res, n = r.p.length, i1 = Math.round((r.first + sec - r.t0) * r.fs), a = 0, b = 0;
    for (var i = 0; i < n; i++) { var v = r.p[i] * r.p[i]; b += v; if (i < i1) a += v; }
    return b > 0 ? a / b : 0;
  }
  function late(sec) {                                        // and in its last `sec` seconds
    var r = st.res, n = r.p.length, i0 = Math.round((r.last - sec - r.t0) * r.fs), a = 0, b = 0;
    for (var i = 0; i < n; i++) { var v = r.p[i] * r.p[i]; b += v; if (i >= i0) a += v; }
    return b > 0 ? a / b : 0;
  }
  function extent() { var e = [Infinity, -Infinity, Infinity, -Infinity]; for (var i = 0; i < st.pts.length; i++) { var q = st.pts[i]; if (q[0] < e[0]) e[0] = q[0]; if (q[0] > e[1]) e[1] = q[0]; if (q[1] < e[2]) e[2] = q[1]; if (q[1] > e[3]) e[3] = q[1]; } return e; }
  window.THUNDERSOUND = function (every) {                    // the samples handed to the loudspeaker, thinned
    var x = samples(), out = [];
    for (var i = 0; i < x.length; i += every) out.push(x[i]);
    return out;
  };
  window.THUNDEREAR = function () {                           // what is passing to the loudspeaker now, and the time since the flash
    if (!st.ear) return { t: 0, loud: 0 };
    var d = new Float32Array(st.ear.fftSize), m = 0; st.ear.getFloatTimeDomainData(d);
    for (var i = 0; i < d.length; i++) if (Math.abs(d[i]) > m) m = Math.abs(d[i]);
    return { t: st.ctx.currentTime - st.tFlash, loud: m };
  };
  window.THUNDERSEEK = function (t) { st.t = t; drawScene(); drawTrace(); };

  var fitting = 0;
  window.addEventListener('resize', function () { clearTimeout(fitting); fitting = setTimeout(function () { fit(); drawScene(); drawTrace(); }, 60); });
  fit();
  rebuild();
})();
